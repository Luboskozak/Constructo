// Constructo (Řízení prací Hybernská) – sdílené úložiště dat a fotek z mobilu (Google Apps Script), verze skriptu 13
// Data se ukládají do souboru rizeni-praci-hybernska-data.json na Disku Google; každý den se uloží i záložní kopie.
const HESLO = '@@HESLO@@';
const SKRIPT_VERZE = 13;

function doGet(e) { return odpoved_(e.parameter.key, Number(e.parameter.rev), null); }
function doPost(e) {
  // chyba uvnitř skriptu se vrátí aplikaci jako text (jinak Google pošle jen stránku „Nezdařilo se“ bez důvodu)
  try { return doPost_(e); } catch (x) { return json_({error: 'Google: ' + ((x && x.message) || String(x))}); }
}
function doPost_(e) {
  let b = {};
  try { b = JSON.parse(e.postData.contents); } catch (x) {}
  if (b.akce === 'dok_start') return dokStart_(b);
  if (b.akce === 'dok_cast') return dokCast_(b);
  if (b.akce === 'dok_cti') return dokCti_(b);
  if (b.akce === 'dok_smaz') return dokSmaz_(b);
  if (b.akce === 'foto' || b.akce === 'mail' || b.akce === 'zapis') {
    const lock = LockService.getScriptLock();
    lock.waitLock(25000);
    try { return b.akce === 'foto' ? foto_(b) : b.akce === 'zapis' ? zapis_(b) : mail_(b); } finally { lock.releaseLock(); }
  }
  rpZapnout_();
  if (b.akce === 'verze') return json_({verze: SKRIPT_VERZE, samo: true});
  if (b.akce === 'samo_aktualizace') return samoAktualizace_(b);
  if (b.akce === 'prehled') return prehled_(b, false);
  if (b.akce === 'prehled_ok') return prehled_(b, true);
  if (b.akce === 'prehled_stav') return prehledStav_(b);
  if (b.akce === 'prehled_test') return prehledTest_(b);
  if (b.akce === 'odborne') return odborne_(b);
  if (b.akce === 'korektura') return korektura_(b);
  if (b.akce === 'formulace') return formulace_(b);
  if (b.akce === 'zapis_pdf') return zapisPdf_(b);
  if (b.akce === 'zavada') return zavada_(b);
  if (b.akce === 'pozvanka') return pozvanka_(b);
  if (b.akce === 'pripojit') return pripojit_(b);
  if (b.akce === 'ares') return ares_(b);
  if (b.akce === 'firma_zprava') return firmaZprava_(b);
  if (b.akce === 'firma_odpoved') return firmaOdpoved_(b);
  if (b.akce === 'kolega_zprava') return kolegaZprava_(b);
  if (b.akce === 'kolega_odpoved') return kolegaOdpoved_(b);
  return odpoved_(b.key, Number(b.rev), b.data || null);
}
// Fotky z mobilní aplikace: složka „Constructo – fotky ze stavby“ na Disku, podsložky po dnech.
// Fotku uvidí každý, kdo má její odkaz (odkaz je jen v datech stavby a v e-mailu).
function foto_(b) {
  if (b.key !== HESLO) return json_({error: 'heslo'});
  const den = /^\d{4}-\d{2}-\d{2}$/.test(b.den || '') ? b.den : Utilities.formatDate(new Date(), 'Europe/Prague', 'yyyy-MM-dd');
  const sl = slozka_(slozka_(DriveApp.getRootFolder(), 'Constructo – fotky ze stavby'), den);
  const jmeno = String(b.name || 'foto.jpg').replace(/[^\w.-]/g, '_');
  const it = sl.getFilesByName(jmeno);
  const f = it.hasNext() ? it.next() : sl.createFile(Utilities.newBlob(Utilities.base64Decode(b.data), b.mime || 'image/jpeg', jmeno));
  try { f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (x) {}
  const id = f.getId();
  return json_({id: id, url: 'https://drive.google.com/file/d/' + id + '/view', thumb: 'https://drive.google.com/thumbnail?id=' + id + '&sz=w800'});
}
// 13: Dokumentace – velké soubory (PDF, DWG) se nahrávají po dílech přímo na Disk Google (složka „Constructo – dokumentace“
// se stejnými podsložkami jako v aplikaci) a po dílech se i stahují. Používá Drive API přes UrlFetch (oprávnění drive už skript má).
function dokSlozka_(cesta) {
  let s = slozka_(DriveApp.getRootFolder(), 'Constructo – dokumentace');
  (Array.isArray(cesta) ? cesta : []).slice(0, 30).forEach(n => { n = String(n || '').trim().slice(0, 200); if (n) s = slozka_(s, n); });
  return s;
}
function dokStart_(b) {
  if (b.key !== HESLO) return json_({error: 'heslo'});
  const sl = dokSlozka_(b.cesta), mime = String(b.mime || 'application/octet-stream');
  const meta = {name: String(b.nazev || 'soubor').slice(0, 250), parents: [sl.getId()], mimeType: mime};
  const r = UrlFetchApp.fetch('https://www.googleapis.com/upload/drive/v3/files?uploadType=resumable&supportsAllDrives=true&fields=id', {
    method: 'post', contentType: 'application/json; charset=UTF-8', payload: JSON.stringify(meta), muteHttpExceptions: true,
    headers: {Authorization: 'Bearer ' + ScriptApp.getOAuthToken(), 'X-Upload-Content-Type': mime, 'X-Upload-Content-Length': String(Number(b.vel) || 0)}});
  if (r.getResponseCode() !== 200) { const t = r.getContentText(); if (r.getResponseCode() === 403 && /accessNotConfigured|has not been used|is disabled/.test(t)) dokZapnoutDrive_(); return json_({error: 'Disk Google ' + r.getResponseCode() + ': ' + t.slice(0, 300)}); }
  const h = r.getHeaders(), loc = h.Location || h.location;
  if (!loc) return json_({error: 'Disk Google nevrátil adresu pro nahrání.'});
  const up = Utilities.getUuid();
  CacheService.getScriptCache().put('dokup:' + up, loc, 21600);
  return json_({up: up});
}
function dokCast_(b) {
  if (b.key !== HESLO) return json_({error: 'heslo'});
  const loc = CacheService.getScriptCache().get('dokup:' + b.up);
  if (!loc) return json_({restart: true});
  const od = Number(b.od) || 0, celkem = Number(b.celkem) || 0, data = b.data ? Utilities.base64Decode(b.data) : [];
  const o = {method: 'put', muteHttpExceptions: true, followRedirects: false, contentType: 'application/octet-stream', headers: {}};
  if (data.length) { o.payload = data; o.headers['Content-Range'] = 'bytes ' + od + '-' + (od + data.length - 1) + '/' + celkem; }
  else o.headers['Content-Range'] = 'bytes */' + celkem;
  const r = UrlFetchApp.fetch(loc, o), kod = r.getResponseCode();
  if (kod === 308) { const rg = String(r.getHeaders().Range || r.getHeaders().range || ''); const m = rg.split('-')[1]; return json_({ok: true, dalsi: m ? Number(m) + 1 : 0}); }
  if (kod === 200 || kod === 201) {
    const id = JSON.parse(r.getContentText()).id;
    try { DriveApp.getFileById(id).setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (x) {}
    CacheService.getScriptCache().remove('dokup:' + b.up);
    return json_({id: id, url: 'https://drive.google.com/file/d/' + id + '/view'});
  }
  if (kod === 404 || kod === 410) return json_({restart: true});
  return json_({error: 'Disk Google ' + kod + ': ' + r.getContentText().slice(0, 300)});
}
function dokCti_(b) {
  const ro = !!b.key && b.key === PropertiesService.getScriptProperties().getProperty('cteni');
  if (b.key !== HESLO && !ro) return json_({error: 'heslo'});
  const id = String(b.id || '').replace(/[^A-Za-z0-9_-]/g, '');
  const r = UrlFetchApp.fetch('https://www.googleapis.com/drive/v3/files/' + id + '?alt=media&supportsAllDrives=true', {
    muteHttpExceptions: true, headers: {Authorization: 'Bearer ' + ScriptApp.getOAuthToken(), Range: 'bytes=' + (Number(b.od) || 0) + '-' + (Number(b.do) || 0)}});
  const kod = r.getResponseCode();
  if (kod !== 200 && kod !== 206) return json_({error: 'Disk Google ' + kod + ': ' + r.getContentText().slice(0, 200)});
  const cr = String(r.getHeaders()['Content-Range'] || r.getHeaders()['content-range'] || ''), bytes = r.getContent();
  return json_({data: Utilities.base64Encode(bytes), celkem: cr ? Number(cr.split('/')[1]) : bytes.length});
}
// Když Google hlásí, že Drive API není pro skript zapnuté, zapne ho skript sám (přidá službu Drive do appsscript.json přes Apps Script API).
function dokZapnoutDrive_() {
  const c = CacheService.getScriptCache();
  if (c.get('dokdrive')) return; c.put('dokdrive', '1', 21600);
  try {
    const api = 'https://script.googleapis.com/v1/projects/' + ScriptApp.getScriptId();
    const vol = (metoda, cesta, telo) => {
      const o = {method: metoda, headers: {Authorization: 'Bearer ' + ScriptApp.getOAuthToken()}, muteHttpExceptions: true};
      if (telo) { o.contentType = 'application/json'; o.payload = JSON.stringify(telo); }
      const x = UrlFetchApp.fetch(api + cesta, o);
      if (x.getResponseCode() >= 300) throw new Error('API ' + x.getResponseCode());
      return x.getContentText() ? JSON.parse(x.getContentText()) : {};
    };
    const soubory = (vol('get', '/content').files || []).map(f => ({name: f.name, type: f.type, source: f.source}));
    const man = soubory.filter(f => f.type === 'JSON' && f.name === 'appsscript')[0];
    if (!man) return;
    const m = JSON.parse(man.source), dep = m.dependencies || {}, sl = dep.enabledAdvancedServices || [];
    if (sl.some(x => x.serviceId === 'drive')) return;
    sl.push({userSymbol: 'Drive', serviceId: 'drive', version: 'v3'}); dep.enabledAdvancedServices = sl; m.dependencies = dep;
    man.source = JSON.stringify(m, null, 2);
    vol('put', '/content', {files: soubory});
  } catch (x) {}
}
function dokSmaz_(b) {
  if (b.key !== HESLO) return json_({error: 'heslo'});
  let n = 0;
  (Array.isArray(b.ids) ? b.ids : []).slice(0, 500).forEach(id => { try { DriveApp.getFileById(String(id)).setTrashed(true); n++; } catch (x) {} });
  return json_({ok: true, smazano: n});
}
// Zvací kódy: administrátor v programu vytvoří krátký kód pro kolegu, aplikace ho tady vymění za připojení.
// Kód platí 14 dní a dá se použít nejvýš na 20 zařízeních.
function pozvanka_(b) {
  if (b.key !== HESLO) return json_({error: 'heslo'});
  const p = PropertiesService.getScriptProperties(), abc = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
  let kod = '';
  do { kod = ''; for (let i = 0; i < 6; i++) kod += abc.charAt(Math.floor(Math.random() * abc.length)); } while (p.getProperty('poz:' + kod));
  const exp = Date.now() + 14 * 864e5;
  p.setProperty('poz:' + kod, JSON.stringify({jmeno: String(b.jmeno || ''), exp: exp, n: 0, role: b.role === 'cteni' ? 'cteni' : ''}));
  return json_({kod: kod, exp: exp});
}
function pripojit_(b) {
  const c = CacheService.getScriptCache(), spatne = Number(c.get('spatne') || 0);
  if (spatne > 100) return json_({error: 'Příliš mnoho pokusů, zkuste to za hodinu.'});
  const kod = String(b.kod || '').toUpperCase().replace(/[^A-Z0-9]/g, '');
  const p = PropertiesService.getScriptProperties(), raw = kod.length === 6 ? p.getProperty('poz:' + kod) : null;
  const z = raw ? JSON.parse(raw) : null;
  if (!z || z.exp < Date.now() || z.n >= 20) { c.put('spatne', String(spatne + 1), 3600); return json_({error: 'kod'}); }
  z.n++; p.setProperty('poz:' + kod, JSON.stringify(z));
  if (z.role === 'cteni') return json_({key: cteni_(p), jmeno: z.jmeno, role: 'cteni'});
  return json_({key: HESLO, jmeno: z.jmeno});
}
// Klíč jen pro čtení (pro pozvané „jen nahlíží“): data jde stáhnout, ale ne změnit. Smazáním vlastnosti „cteni“ se všem nahlížejícím přístup zruší.
function cteni_(p) {
  let k = p.getProperty('cteni');
  if (!k) { k = 'cteni-' + Utilities.getUuid().replace(/-/g, ''); p.setProperty('cteni', k); }
  return k;
}
function slozka_(rodic, jmeno) { const it = rodic.getFoldersByName(jmeno); return it.hasNext() ? it.next() : rodic.createFolder(jmeno); }
// E-mail vybraným lidem z mobilní aplikace. Posílá jen na adresy uložené u lidí v programu (Upravit osobu → E-mail).
function mail_(b) {
  if (b.key !== HESLO) return json_({error: 'heslo'});
  const p = PropertiesService.getScriptProperties();
  if (p.getProperty('mail:' + b.cid)) return json_({ok: true, opakovane: true});
  const d = nacti_(p) || {}, lide = (d.project && d.project.people) || {};
  const komu = [], chybi = [];
  (b.to || []).forEach(function (n) { const e = lide[n] && lide[n].email; if (e) komu.push(e); else chybi.push(n); });
  if (!komu.length) { p.setProperty('mail:' + b.cid, '0'); return json_({ok: true, sent: 0, missing: chybi}); }
  const esc = function (s) { return String(s).replace(/[&<>"]/g, function (c) { return {'&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;'}[c]; }); };
  const text = String(b.text || '').trim() || 'Foto ze stavby';
  const stavba = (d.project && d.project.name) || 'Stavba';
  const predmet = (b.info ? 'Na vědomí od ' : 'Nový úkol od ') + (b.by || 'kolegy') + ': ' + text.split('\n')[0].slice(0, 70);
  const obr = {}; let fotky = '';
  (b.fotky || []).slice(0, 6).forEach(function (id, i) {
    try { obr['f' + i] = DriveApp.getFileById(id).getBlob(); fotky += '<p><a href="https://drive.google.com/file/d/' + id + '/view"><img src="cid:f' + i + '" style="max-width:100%;border-radius:8px"></a></p>'; } catch (x) {}
  });
  const termin = b.end ? String(b.end).split('-').reverse().map(Number).join('. ') : '';
  const telo = '<div style="font-family:Arial,sans-serif;font-size:15px;color:#16191c">' +
    '<p style="font-size:17px"><b>' + esc(text).replace(/\n/g, '<br>') + '</b></p>' +
    '<p style="color:#555">' + esc(stavba) + (b.obj ? ' · objekt ' + esc(b.obj) : '') + (b.info ? ' · na vědomí' : termin ? ' · splnit do ' + termin : '') +
    '<br>Zadal ' + esc(b.by || '') + ' z mobilní aplikace Constructo</p>' + fotky +
    '<p style="color:#555">' + (b.info ? 'Informaci najdete v programu Constructo v záložce Úkoly.' : 'Úkol najdete v programu Constructo v záložce Úkoly. Až bude hotový, klikněte Splněno.') + '</p></div>';
  MailApp.sendEmail({to: komu.join(','), subject: predmet, htmlBody: telo, inlineImages: obr, name: 'Constructo'});
  p.setProperty('mail:' + b.cid, '1');
  return json_({ok: true, sent: komu.length, missing: chybi});
}
function odpoved_(key, rev, data) {
  const ro = !!key && key !== HESLO && key === PropertiesService.getScriptProperties().getProperty('cteni');
  if (key !== HESLO && !ro) return json_({error: 'heslo'});
  if (ro && data) return json_({error: 'Jen prohlížení: změny se neukládají.'});
  const lock = LockService.getScriptLock();
  lock.waitLock(25000);
  try {
    const p = PropertiesService.getScriptProperties();
    let cur = nacti_(p), r = Number(p.getProperty('rev') || 0);
    if (data) { cur = mergeData(cur || {}, data); r++; uloz_(p, cur); p.setProperty('rev', String(r)); }
    if (!data && rev === r) return json_({rev: r, same: true});
    return json_({rev: r, data: cur || {}});
  } finally { lock.releaseLock(); }
}
function nacti_(p) {
  const id = p.getProperty('fileId');
  return id ? JSON.parse(DriveApp.getFileById(id).getBlob().getDataAsString('UTF-8')) : null;
}
function uloz_(p, d) {
  const s = JSON.stringify(d);
  let id = p.getProperty('fileId');
  if (id) DriveApp.getFileById(id).setContent(s);
  else { id = DriveApp.createFile('rizeni-praci-hybernska-data.json', s, 'application/json').getId(); p.setProperty('fileId', id); }
  const den = Utilities.formatDate(new Date(), 'Europe/Prague', 'yyyy-MM-dd');
  if (p.getProperty('zaloha') !== den) { DriveApp.getFileById(id).makeCopy('zaloha-rizeni-praci-' + den + '.json'); p.setProperty('zaloha', den); }
}
function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); }

function mergeData(a, b) {
    const KEYS = {tasks: r => r.id, hmg: r => String(r.hid), hmgGroups: r => r.obj, diary: r => r.id, log: r => r.id, zavady: r => r.id, dokumenty: r => r.id};
    a = a || {}; b = b || {};
    const out = Object.assign({}, a); out.tomb = {};
    Object.keys(KEYS).forEach(c => {
      const k = KEYS[c], ta = (a.tomb || {})[c] || {}, tb = (b.tomb || {})[c] || {}, tomb = Object.assign({}, ta);
      Object.keys(tb).forEach(x => { if (!(tomb[x] >= tb[x])) tomb[x] = tb[x]; });
      const m = new Map();
      (a[c] || []).concat(b[c] || []).forEach(r => { const key = k(r), o = m.get(key); if (!o || (r._ts || 0) > (o._ts || 0)) m.set(key, r); });
      out[c] = Array.from(m.values()).filter(r => !(tomb[k(r)] >= (r._ts || 0)));
      out.tomb[c] = tomb;
    });
    ['project', 'hmgInfo'].forEach(f => { out[f] = ((b[f] || {})._ts || 0) > ((a[f] || {})._ts || 0) ? b[f] : (a[f] || b[f]); });
    // 4.34: firmy, příjemci, lidé a tým v údajích stavby se slučují po jednom (dřív vyhrála celá novější kopie a starší počítač mohl firmy smazat)
    const pa = a.project, pb = b.project;
    if (pa && pb && pa !== pb) {
      const win = out.project === pb ? pb : pa, los = win === pb ? pa : pb, P = Object.assign({}, win), sm = {};
      ['firmy', 'prijemci', 'people', 'team'].forEach(k => { sm[k] = Object.assign({}, ((los.smazane || {})[k]) || {}); const w = ((win.smazane || {})[k]) || {}; Object.keys(w).forEach(x => { if (!(sm[k][x] >= w[x])) sm[k][x] = w[x]; }); });
      P.smazane = sm;
      ['firmy', 'prijemci'].forEach(k => {
        const m = new Map();
        (Array.isArray(los[k]) ? los[k] : []).forEach(r => { if (r && r.id) m.set(r.id, {r, w: false}); });
        (Array.isArray(win[k]) ? win[k] : []).forEach(r => { if (!r || !r.id) return; const o = m.get(r.id); if (!o || (r._ts || 0) >= (o.r._ts || 0)) m.set(r.id, {r, w: true}); });
        P[k] = Array.from(m.values()).filter(x => x.w || !(sm[k][x.r.id] >= (x.r._ts || 0))).map(x => x.r);
      });
      const pw = win.people || {}, pl = los.people || {}, pp = Object.assign({}, pw);
      Object.keys(pl).forEach(n => { if (!(n in pp) && !(sm.people[n] >= ((pl[n] || {})._ts || 0))) pp[n] = pl[n]; });
      if (win.people || los.people) P.people = pp;
      if (Array.isArray(win.team) || Array.isArray(los.team)) { const t = Array.isArray(win.team) ? win.team.slice() : []; (Array.isArray(los.team) ? los.team : []).forEach(n => { if (!t.includes(n) && !sm.team[n]) t.push(n); }); P.team = t; }
      out.project = P;
    }
    return out;
  }
// Zpráva o závadě pro firmu (z mobilu): napíše ji Claude podle popisu, fotek a výřezu půdorysu.

// Verze 12: reakce kolegy na zprávu v kolektivu (Na vědomí / Úkol) přes odkaz v e-mailu – stránka /reakce/, bez přihlášení a bez kódu.
// Odkaz nese skupinu zprávy a kód = HMAC-SHA256(skupina, HESLO); nikde se neukládá. Stránka ukáže jen tuto jednu zprávu.
function kolegaTok_(grp) { return Utilities.base64EncodeWebSafe(Utilities.computeHmacSha256Signature(String(grp), HESLO)).slice(0, 22); }
function kolegaNajdi_(d, b) {
  const z = String(b.z || ''), t = String(b.t || '');
  if (!z || t.length !== 22 || kolegaTok_(z) !== t) return null;
  const ts = (d.tasks || []).filter(function (x) { return x && x.grp === z; });
  return ts.length ? ts : null;
}
function kolegaVen_(d, ts) {
  const t = ts[0], z = ts.map(function (x) { return x.zprava; }).filter(Boolean)[0] || null;
  const komu = z && z.komu && z.komu.length ? z.komu : ts.reduce(function (a, x) { return a.concat(t.info ? [x.owner].concat(x.team || []) : [x.owner]); }, []).filter(function (n, i, a) { return n && a.indexOf(n) === i; });
  const rk = {}; ts.forEach(function (x) { (x.reakce || []).forEach(function (o) { rk[o.cid || (o.at + o.jmeno)] = o; }); });
  return {stavba: (d.project && d.project.name) || '', info: !!t.info, text: t.title || '', by: t.createdBy || '', at: t.createdAt || '', end: t.info ? '' : (t.end || ''), obj: t.obj || '', komu: komu,
    hotovo: ts.filter(function (x) { return !t.info && (x.done >= 100 || x.status === 'done'); }).map(function (x) { return x.owner; }),

function kolegaOdpoved_(b) {

// plánovač (každý den 6:00–7:00): nastaví jednorázové spuštění ranniPrehled přesně na 7:00
function ranniPlan() {
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'ranniPrehled') ScriptApp.deleteTrigger(t); });
  const sedm = new Date(Utilities.formatDate(new Date(), 'Europe/Prague', "yyyy-MM-dd'T'07:00:00XXX"));
  if (sedm.getTime() - Date.now() < 60000) { ranniPrehled(); return; }
  ScriptApp.newTrigger('ranniPrehled').timeBased().at(sedm).create();
}
function ranniPrehled() {
  ScriptApp.getProjectTriggers().forEach(function (t) { if (t.getHandlerFunction() === 'ranniPrehled') ScriptApp.deleteTrigger(t); });
  const p = PropertiesService.getScriptProperties(), d = nacti_(p) || {}, dnes = rpDnes_(), wd = rpDenTydne_(dnes), pondeli = rpPlus_(dnes, -wd);
  if (wd > 4 && !RP_VIKEND) return;
  rpLide_(d).forEach(function (o) {
    try {
      const k = rpKlic_(o.n);
      let tok = p.getProperty(k), st = rpStav_(p, tok);
      const novy = !st || st.w !== pondeli;
      if (!novy && (st.ok || (st.poslano || []).indexOf(dnes) >= 0)) return;
      const r = ranniData(d, o.n, dnes);
      if (!r.po.length && !r.tyden.length) { if (novy) { tok = Utilities.getUuid().replace(/-/g, ''); p.setProperty('rp:' + tok, JSON.stringify({n: o.n, w: pondeli, ok: 'nic', poslano: []})); p.setProperty(k, tok); } return; }
      if (novy) { tok = Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '').slice(0, 8); st = {n: o.n, w: pondeli, ok: '', poslano: []}; }
      rpPosli_(d, o.email, o.n, dnes, tok, !novy, false);
      const lock = LockService.getScriptLock(); lock.waitLock(25000);
      try { const s2 = novy ? st : (rpStav_(p, tok) || st); s2.poslano = (s2.poslano || []).concat([dnes]); p.setProperty('rp:' + tok, JSON.stringify(s2)); if (novy) p.setProperty(k, tok); } finally { lock.releaseLock(); }
    } catch (x) { console.error('Ranní přehled ' + o.n + ': ' + ((x && x.message) || x)); }
  });
  // staré potvrzovací odkazy (starší než 6 týdnů) uklidit
  const vse = p.getProperties(), hranice = rpPlus_(pondeli, -42);
  Object.keys(vse).forEach(function (k) { if (k.indexOf('rp:') === 0) { try { if (JSON.parse(vse[k]).w < hranice) p.deleteProperty(k); } catch (x) {} } });
}

    const sablona = r.getContentText();
    const i = sablona.indexOf('const SKRIPT_VERZE = ');

    const vol = (metoda, cesta, telo) => {
      const o = {method: metoda, headers: {Authorization: 'Bearer ' + ScriptApp.getOAuthToken()}, muteHttpExceptions: true};
      if (telo) { o.contentType = 'application/json'; o.payload = JSON.stringify(telo); }

      const t = x.getContentText();
      if (x.getResponseCode() >= 300) throw new Error('API ' + x.getResponseCode() + ': ' + t.slice(0, 300));
      return t ? JSON.parse(t) : {};
    };
    const obsah = vol('get', '/content');
    const soubory = (obsah.files || []).map(f => ({name: f.name, type: f.type, source: f.source}));
    const hlavni = soubory.filter(f => f.type === 'SERVER_JS' && String(f.source).indexOf('const HESLO') >= 0)[0] || soubory.filter(f => f.type === 'SERVER_JS')[0];
    if (!hlavni) return json_({ok: false, duvod: 'soubor'});
    hlavni.source = kod;
    vol('put', '/content', {files: soubory});
    const verze = vol('post', '/versions', {description: 'Constructo skript ' + nova});
    const moje = String(ScriptApp.getService().getUrl() || '');
    const nasazeni = (vol('get', '/deployments').deployments || []).filter(d => d.deploymentConfig && d.deploymentConfig.versionNumber && (d.entryPoints || []).some(p => p.entryPointType === 'WEB_APP'));
    const cil = nasazeni.filter(d => moje.indexOf(d.deploymentId) >= 0);
    const vybrane = cil.length ? cil : nasazeni.length === 1 ? nasazeni : [];
    if (!vybrane.length) return json_({ok: false, duvod: 'nasazeni', verze: SKRIPT_VERZE});
    vybrane.forEach(d => vol('put', '/deployments/' + d.deploymentId, {deploymentConfig: {scriptId: id, versionNumber: verze.versionNumber, manifestFileName: 'appsscript', description: 'Constructo skript ' + nova}}));
    return json_({ok: true, verze: nova, predtim: SKRIPT_VERZE});
  } catch (x) {
    return json_({ok: false, duvod: String((x && x.message) || x).slice(0, 400)});
  } finally { lock.releaseLock(); }
}

function ranniData(d, jmeno, dnes) {
    const add = (s, n) => { const p = String(s).split('-').map(Number), x = new Date(Date.UTC(p[0], p[1] - 1, p[2] + n)); return x.toISOString().slice(0, 10); };
    const doK = add(dnes, 7), out = [];
    (d.tasks || []).forEach(t => {
      if (!t || t.info) return;
      const hot = t.status === 'done' || (Number(t.done) || 0) >= 100, lidi = [t.owner].concat(t.team || []).filter((n, i, a) => n && a.indexOf(n) === i);
      if (!hot && t.end && lidi.indexOf(jmeno) >= 0) out.push({end: t.end, text: String(t.title || '(bez názvu)'), obj: t.obj || '', pct: Number(t.done) || 0, spolu: lidi.filter(n => n !== jmeno), sled: t.kind !== 'simple'});
      (t.sub || []).forEach(u => {
        if (hot || !u || u.done || !u.end || !String(u.text || '').trim() || (u.owner || t.owner) !== jmeno) return;
        out.push({end: u.end, text: String(u.text), ukol: String(t.title || ''), obj: t.obj || '', dilci: true, spolu: []});
      });
    });
    out.sort((a, b) => a.end.localeCompare(b.end) || a.text.localeCompare(b.text, 'cs'));
    return {po: out.filter(x => x.end < dnes), tyden: out.filter(x => x.end >= dnes && x.end <= doK), od: dnes, do: doK};
  }

        '<td style="' + td + '">' + popis(x) + '</td><td style="' + td + 'color:#6b7279">' + e(x.obj) + '</td></tr>').join('') + '</table>';

      cxSekce('2. Úkoly na příští týden · ' + dt(r.od) + ' – ' + dt(r.do) + ' (' + r.tyden.length + ')') + (r.tyden.length ? tab(r.tyden, false) : prazdne('Na příští týden nemáte žádný úkol s termínem.')) +
      (proMail ? '' : '<p style="margin:16px 0 0;font-size:12.5px;color:#6b7279">Hotové úkoly označte v Constructu v záložce Úkoly tlačítkem Splněno. Pokud termín nestihnete, domluvte nový termín se stavbyvedoucím.</p>');
